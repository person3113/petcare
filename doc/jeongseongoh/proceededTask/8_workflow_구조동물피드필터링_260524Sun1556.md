  - 예시 코드 뼈대:
    ```js
    const debounceRef = useRef(null);

    function handleFilterChange(newFilters) {
      clearTimeout(debounceRef.current);
      debounceRef.current = setTimeout(() => {
        applyFilter(newFilters);
      }, 300);
    }
    ```

### 4단계: Array.filter 체이닝 로직 구현

- [x] `AnimalFeedPage.jsx` 안에 **`applyFilter(animals, filters)`** 함수 작성
  - 순서대로 조건 체이닝:
    1. 시도 필터: `animal.jurisdiction.includes(filters.sido)`
    2. 시군구 필터: `animal.jurisdiction.includes(filters.sigungu)`
    3. 보호소 필터: `animal.shelterName === filters.shelterName`
    4. 축종 필터: `animal.kind.includes(filters.kind)` (예: `[개]`, `[고양이]`)
    5. 상태 필터(`status`): `animal.status === filters.status` (예: `보호중`)
    6. 성별 필터(`gender`): `animal.gender === filters.gender`
    7. 중성화 필터(`isNeutered`): `animal.isNeutered === filters.isNeutered`
  - 각 필터 값이 `''`(전체) 이면 해당 조건 건너뜀 (skip)
  - **sfeSoci 후처리**: `socialization` 필드가 존재하고 비어있지 않은 데이터만 포함 (선택 옵션)
  - **sfeHealth 후처리**: `healthStatus === '양호'` 데이터만 포함 (선택 옵션)
  - 예시 코드 뼈대:
    ```js
    function applyFilter(animals, filters) {
      let result = animals;

      if (filters.sido) {
        result = result.filter(a => a.jurisdiction.includes(filters.sido));
      }
      if (filters.sigungu) {
        result = result.filter(a => a.jurisdiction.includes(filters.sigungu));
      }
      if (filters.shelterName) {
        result = result.filter(a => a.shelterName === filters.shelterName);
      }
      if (filters.kind) {
        result = result.filter(a => a.kind.includes(filters.kind));
      }
      if (filters.status) {
        result = result.filter(a => a.status === filters.status);
      }
      if (filters.gender) {
        result = result.filter(a => a.gender === filters.gender);
      }
      if (filters.isNeutered) {
        result = result.filter(a => a.isNeutered === filters.isNeutered);
      }
      // sfeSoci 후처리
      if (filters.onlySocialized) {
        result = result.filter(a => a.socialization && a.socialization.trim() !== '');
      }
      // sfeHealth 후처리
      if (filters.onlyHealthy) {
        result = result.filter(a => a.healthStatus === '양호');
      }

      return result;
    }
    ```
